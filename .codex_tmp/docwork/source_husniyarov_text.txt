0285	Normal	2.1 Описание программы
0286	Normal	В рамках разработки программного продукта реализовано единое мобильное приложение Liga Gym, предназначенное для авторизации пользователя, учета тренировок, маршрутов, питания, шаговой активности, целей, аналитики прогресса, общения с друзьями, работы с тренером и получения AI-рекомендаций по фитнесу.
0287	Normal	Клиентская часть реализована на языке Dart с использованием фреймворка Flutter и библиотеки Material 3. 1453	Normal	static double calculateDistanceBetween(
1454	Normal	WorkoutRoutePoint start,
1455	Normal	WorkoutRoutePoint end,
1456	Normal	) {
1457	Normal	if (!start.hasValidCoordinates || !end.hasValidCoordinates) {
1458	Normal	return 0;
1459	Normal	}
1461	Normal	const earthRadiusMeters = 6371000.0;
1462	Normal	final latitudeDistance = _toRadians(end.latitude - start.latitude);
1463	Normal	final longitudeDistance = _toRadians(end.longitude - start.longitude);
1464	Normal	final startLatitude = _toRadians(start.latitude);
1465	Normal	final endLatitude = _toRadians(end.latitude);
1467	Normal	final haversine =
1468	Normal	math.pow(math.sin(latitudeDistance / 2), 2) +
1469	Normal	math.cos(startLatitude) *
1470	Normal	math.cos(endLatitude) *
1471	Normal	math.pow(math.sin(longitudeDistance / 2), 2);
1472	Normal	final normalizedHaversine = haversine.clamp(0, 1).toDouble();
1473	Normal	final angularDistance =
1474	Normal	2 *
1475	Normal	math.atan2(
1476	Normal	math.sqrt(normalizedHaversine),
1477	Normal	math.sqrt(1 - normalizedHaversine),
1478	Normal	);
1480	Normal	final distance = earthRadiusMeters * angularDistance;
1481	Normal	return distance.isFinite ? distance : 0;
1482	Normal	}
1484	Normal	static WorkoutRouteTelemetry analyzeRouteTelemetry(
1485	Normal	List<WorkoutRoutePoint> route,
1486	Normal	) {
1487	Normal	if (route.length < 2) {
1488	Normal	return const WorkoutRouteTelemetry();
1489	Normal	}
1491	Normal	var elevationGainMeters = 0.0;
1492	Normal	var previousAltitude = route.first.altitudeMeters;
1493	Normal	for (final point in route.skip(1)) {
1494	Normal	final altitude = point.altitudeMeters;
1495	Normal	if (altitude == null || !altitude.isFinite) {
1496	Normal	continue;
1497	Normal	}
1498	Normal	if (previousAltitude == null || !previousAltitude.isFinite) {
1499	Normal	previousAltitude = altitude;
1500	Normal	continue;
1501	Normal	}
1503	Normal	final delta = altitude - previousAltitude;
1504	Normal	if (delta > 1.5) {
1505	Normal	elevationGainMeters += delta;
1506	Normal	}
1507	Normal	previousAltitude = altitude;
1508	Normal	}
1510	Normal	final speedSamples = route
1511	Normal	.map((point) => point.speedMetersPerSecond)
1512	Normal	.whereType<double>()
1513	Normal	.where((speed) => speed.isFinite && speed > 0 && speed < 12)
1514	Normal	.toList(growable: false);

1516	Normal	? 0.0
1517	Normal	: speedSamples.reduce((left, right) => left + right) /
1518	Normal	speedSamples.length;

1521	Normal	.map((point) => point.accuracyMeters)
1522	Normal	.whereType<double>()
1523	Normal	.where((accuracy) => accuracy.isFinite && accuracy > 0)
1524	Normal	.toList(growable: false);
1525	Normal	final averageAccuracy = accuracySamples.isEmpty
1526	Normal	? null
1527	Normal	: accuracySamples.reduce((left, right) => left + right) /
1528	Normal	accuracySamples.length;
1530	Normal	return WorkoutRouteTelemetry(
1531	Normal	elevationGainMeters: elevationGainMeters,
1532	Normal	averageSpeedMetersPerSecond: averageSpeed,
1533	Normal	averageAccuracyMeters: averageAccuracy,

1535	Normal	);
